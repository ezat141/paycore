# PayCore M3 Implementation Plan

**Goal:** Every committed payment status change is written to a transactional outbox in the same database transaction. A background relay publishes it to Kafka at least once. A new `history-service` stores it idempotently and serves it through a secured read API. The M0–M2 no-double-charge and no-lost-charge guarantees are left untouched.

**Architecture:**
- payment-service gains an `outbox_events` table, written by a `PaymentEvents` component that joins the existing `PaymentWrites` transactions.
- An `OutboxRelay` (scheduled, never on a request path) claims rows with `FOR UPDATE SKIP LOCKED`, publishes them keyed by payment id, and marks them published only after the broker acks.
- `history-service` is a new module with its own Postgres. It consumes with `INSERT … ON CONFLICT (event_id) DO NOTHING`, orders by `version`, sends poison records to a DLT, and retries database outages without limit.

**Tech stack:** Java 21, Spring Boot 4.1.0, Spring Kafka 4.1.0 (kafka-clients 4.2.1), Kafka `apache/kafka:4.2.0` (KRaft), PostgreSQL 16, Flyway, Spring JDBC (`JdbcClient`), Testcontainers 2.0.5 (`testcontainers-kafka`), Awaitility (from the test starter).

**Spec:** `docs/specs/2026-10-03-paycore-m3-design.md`. Section references (§5.1 etc.) point into it.

```sql
-- The transactional outbox (spec §4.2). A row is written in the same transaction as the payment
-- status change it describes, so "change committed" and "event recorded" cannot disagree.
CREATE TABLE outbox_events (
    id           uuid         PRIMARY KEY,           -- the eventId
    -- Relay order. For one payment, seq order is commit order: the next change can only insert
    -- after reading the previous change's committed version (optimistic locking, spec §5.1).
    seq          bigserial    NOT NULL UNIQUE,
    aggregate_id uuid         NOT NULL,              -- payment id, the Kafka key
    event_type   varchar(64)  NOT NULL,
    payload      jsonb        NOT NULL,
    created_at   timestamptz  NOT NULL DEFAULT now(),
    published_at timestamptz                          -- set only after the broker acknowledges
);

-- The relay only ever scans unpublished rows, so it stays cheap however large the table grows.
CREATE INDEX outbox_events_unpublished_idx ON outbox_events (seq) WHERE published_at IS NULL;
```

```java
package com.paycore.payment.outbox;

import com.paycore.payment.payment.Payment;
import com.paycore.payment.payment.PaymentRepository;
import com.paycore.payment.payment.PaymentStatus;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import tools.jackson.databind.json.JsonMapper;

import java.util.UUID;

/**
 * Writes the outbox row for a payment status change (spec §4.3), inside the caller's
 * transaction. {@code MANDATORY}: it must never commit on its own, or an event could outlive
 * a change that rolled back.
 */
@Component
public class PaymentEvents {

    /** Its own mapper: the payload is a wire contract and must not follow Boot's global Jackson settings. */
    private static final JsonMapper JSON = JsonMapper.builder().build();

    private final PaymentRepository payments;
    private final JdbcClient jdbc;

    public PaymentEvents(PaymentRepository payments, JdbcClient jdbc) {
        this.payments = payments;
        this.jdbc = jdbc;
    }

    /** Writes an event when, and only when, the status changed from {@code from}. */
    @Transactional(propagation = Propagation.MANDATORY)
    public void statusChanged(Payment payment, PaymentStatus from) {
        if (payment.getStatus() == from) {
            return;
        }
        // @Version is incremented at flush, so the version is read only after one (spec §4.3).
        // Flushing through the repository proxy keeps exception translation: a lost optimistic
        // lock still surfaces as Spring's OptimisticLockingFailureException, which callers handle.
        payments.flush();
        PaymentStatusChanged event = new PaymentStatusChanged(UUID.randomUUID(), payment.getId(),
                payment.getTenant(), payment.getVersion(), from, payment.getStatus(),
                payment.getAmountMinor(), payment.getCurrency(), payment.getDeclineCode(),
                payment.getUpdatedAt());
        jdbc.sql("""
                        INSERT INTO outbox_events (id, aggregate_id, event_type, payload)
                        VALUES (:id, :aggregateId, :type, CAST(:payload AS jsonb))""")
```

```java
package com.paycore.payment.outbox;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Repository;

import java.util.List;
import java.util.UUID;

/** The relay's view of {@code outbox_events}. Both methods run inside the relay's transaction. */
@Repository
public class OutboxStore {

    private final JdbcClient jdbc;

    public OutboxStore(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    /**
     * Claims the oldest unpublished rows in {@code seq} order. {@code SKIP LOCKED} lets two passes,
     * or two instances, take disjoint rows instead of waiting on each other.
     */
    public List<OutboxRow> claimUnpublished(int limit) {
        return jdbc.sql("""
                        SELECT id, seq, aggregate_id, event_type, payload::text AS payload
                        FROM outbox_events
                        WHERE published_at IS NULL
                        ORDER BY seq
                        LIMIT :limit
                        FOR UPDATE SKIP LOCKED""")
                .param("limit", limit)
                .query((rs, row) -> new OutboxRow(
                        rs.getObject("id", UUID.class),
                        rs.getLong("seq"),
                        rs.getObject("aggregate_id", UUID.class),
                        rs.getString("event_type"),
                        rs.getString("payload")))
                .list();
    }

    public void markPublished(List<UUID> ids) {
        if (ids.isEmpty()) {
            return;
        }
        jdbc.sql("UPDATE outbox_events SET published_at = now() WHERE id IN (:ids)")
                .param("ids", ids)
                .update();
    }
}
```

```java
package com.paycore.payment.outbox;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;

import java.util.ArrayList;
import java.util.List;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

/**
 * Publishes outbox rows to Kafka at least once (spec §5.1). Never on a request path: payments
 * keep working while Kafka is down, and the backlog drains when it returns.
 *
 * <p>Each pass claims rows in {@code seq} order, sends them all, then waits for acknowledgements
 * oldest first and marks only the rows up to the first failure. A row that was sent but not
 * acknowledged stays unpublished and is sent again: duplicates are possible by design, and the
 * consumer absorbs them. The pass's transaction stays open while its batch is in flight, bounded
 * by the producer's delivery timeout (spec §5.2).
 */
@Component
public class OutboxRelay {

    private static final Logger log = LoggerFactory.getLogger(OutboxRelay.class);

    private final OutboxStore store;
    private final EventPublisher publisher;
    private final OutboxProperties properties;
    private final TransactionTemplate transaction;

    public OutboxRelay(OutboxStore store, EventPublisher publisher, OutboxProperties properties,
                       PlatformTransactionManager transactionManager) {
        this.store = store;
        this.publisher = publisher;
        this.properties = properties;
        this.transaction = new TransactionTemplate(transactionManager);
    }

    @Scheduled(fixedDelayString = "${paycore.outbox.poll-interval-ms:500}")
    void scheduledPass() {
        if (properties.relayEnabled()) {
            relayOnce();
        }
    }

    /** One pass. Returns how many rows the broker acknowledged and were marked published. */
    public int relayOnce() {
        Integer published = transaction.execute(status -> {
            List<OutboxRow> rows = store.claimUnpublished(properties.batchSize());
            if (rows.isEmpty()) {
                return 0;
            }
            List<CompletableFuture<?>> sends = new ArrayList<>();
            for (OutboxRow row : rows) {
                try {
                    sends.add(publisher.publish(row));
                } catch (RuntimeException e) {
                    log.warn("outbox send of event {} failed immediately: {}", row.id(), e.toString());
                    break;
                }
            }
            List<UUID> acknowledged = new ArrayList<>();
            for (int i = 0; i < sends.size(); i++) {
                try {
                    sends.get(i).get(properties.sendTimeout().toMillis(), TimeUnit.MILLISECONDS);
                    acknowledged.add(rows.get(i).id());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                } catch (ExecutionException | TimeoutException e) {
                    log.warn("outbox event {} was not acknowledged; it stays unpublished: {}",
                            rows.get(i).id(), e.toString());
                    break;
                }
            }
            store.markPublished(acknowledged);
            return acknowledged.size();
        });
        return published == null ? 0 : published;
    }
}
```

```sql
-- Append-only log of payment status changes (spec §6.2). The event id is the primary key, so
-- storing an event again is a no-op: that is what makes at-least-once delivery harmless.
CREATE TABLE payment_history (
    event_id     uuid         PRIMARY KEY,
    payment_id   uuid         NOT NULL,
    tenant       varchar(100) NOT NULL,
    version      bigint       NOT NULL,
    from_status  varchar(16),
    to_status    varchar(16)  NOT NULL CHECK (to_status IN ('PENDING', 'AUTHORIZED', 'DECLINED', 'UNKNOWN',
                                                            'CAPTURING', 'CAPTURED', 'VOIDING', 'VOIDED')),
    amount       bigint       NOT NULL,
    currency     char(3)      NOT NULL,
    decline_code varchar(64),
    occurred_at  timestamptz  NOT NULL,
    received_at  timestamptz  NOT NULL DEFAULT now(),
    UNIQUE (payment_id, version)
);

CREATE INDEX payment_history_read_idx ON payment_history (tenant, payment_id, version);
```

`history/HistoryStore.java`:

```java
package com.paycore.history.history;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Repository;

import java.sql.Types;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.List;
import java.util.UUID;

@Repository
public class HistoryStore {

    private final JdbcClient jdbc;

    public HistoryStore(JdbcClient jdbc) {
        this.jdbc = jdbc;
    }

    /** Stores the event once. Returns false when it was already stored (a redelivery). */
    public boolean insertIfAbsent(PaymentStatusChanged event) {
        return jdbc.sql("""
                        INSERT INTO payment_history (event_id, payment_id, tenant, version, from_status,
                                                     to_status, amount, currency, decline_code, occurred_at)
                        VALUES (:eventId, :paymentId, :tenant, :version, :fromStatus,
                                :toStatus, :amount, :currency, :declineCode, :occurredAt)
                        ON CONFLICT (event_id) DO NOTHING""")
                .param("eventId", event.eventId())
                .param("paymentId", event.paymentId())
                .param("tenant", event.tenant())
                .param("version", event.version())
                .param("fromStatus", event.fromStatus(), Types.VARCHAR)
                .param("toStatus", event.toStatus())
                .param("amount", event.amount())
                .param("currency", event.currency())
                .param("declineCode", event.declineCode(), Types.VARCHAR)
                .param("occurredAt", OffsetDateTime.ofInstant(event.occurredAt(), ZoneOffset.UTC))
                .update() == 1;
    }

    /** One tenant's view of one payment, in version order. Another tenant's payment is empty. */
    public List<HistoryEntry> find(String tenant, UUID paymentId) {
        return jdbc.sql("""
                        SELECT version, from_status, to_status, decline_code, occurred_at
                        FROM payment_history
                        WHERE tenant = :tenant AND payment_id = :paymentId
                        ORDER BY version""")
                .param("tenant", tenant)
                .param("paymentId", paymentId)
                .query((rs, row) -> new HistoryEntry(
                        rs.getLong("version"),
                        rs.getString("from_status"),
                        rs.getString("to_status"),
                        rs.getString("decline_code"),
                        rs.getObject("occurred_at", OffsetDateTime.class).toInstant()))
                .list();
    }
}
```

```java
package com.paycore.history.history;

import org.springframework.stereotype.Component;
import tools.jackson.core.JacksonException;
import tools.jackson.databind.DeserializationFeature;
import tools.jackson.databind.json.JsonMapper;

import java.util.Set;

/**
 * Turns a record's value into an event, or rejects it as invalid. Records are read as strings
 * and parsed here rather than by a Kafka deserializer, so a malformed record becomes one
 * classified exception instead of a deserialization loop. Unknown fields are ignored, so a
 * producer may add fields before this consumer knows them.
 */
@Component
public class EventParser {

    private static final Set<String> STATUSES = Set.of("PENDING", "AUTHORIZED", "DECLINED", "UNKNOWN",
            "CAPTURING", "CAPTURED", "VOIDING", "VOIDED");

    private static final JsonMapper JSON = JsonMapper.builder()
            .disable(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES)
            .build();

    public PaymentStatusChanged parse(String payload) {
        PaymentStatusChanged event;
        try {
            event = JSON.readValue(payload, PaymentStatusChanged.class);
        } catch (JacksonException e) {
            throw new InvalidEventException("unreadable event", e);
        }
        if (event == null || event.eventId() == null || event.paymentId() == null || event.occurredAt() == null) {
            throw new InvalidEventException("event is missing an id or a timestamp");
        }
        if (event.tenant() == null || event.tenant().isBlank()) {
            throw new InvalidEventException("event has no tenant");
        }
        if (event.version() < 0) {
            throw new InvalidEventException("negative version");
        }
        if (!STATUSES.contains(event.toStatus())
                || (event.fromStatus() != null && !STATUSES.contains(event.fromStatus()))) {
            throw new InvalidEventException("unknown status");
        }
        if (event.amount() <= 0 || event.currency() == null || !event.currency().matches("[A-Z]{3}")) {
            throw new InvalidEventException("invalid amount or currency");
        }
        return event;
    }
}
```

```java
/**
 * Topics and failure handling (spec §6.4). An invalid record goes straight to the DLT, so it
 * cannot stall its partition; anything else (a database outage) is retried without limit,
 * because the event is valid and sending it to the DLT would lose history.
 */
@Configuration
public class KafkaConfig {

    public static final String TOPIC = "paycore.payment-events";
    public static final String DLT = "paycore.payment-events.DLT";

    /** Declared here too: whichever service starts first creates it, with the same shape. */
    @Bean
    NewTopic paymentEventsTopic(@Value("${paycore.history.topic-partitions:3}") int partitions,
                                @Value("${paycore.history.topic-replicas:1}") int replicas) {
        return TopicBuilder.name(TOPIC).partitions(partitions).replicas(replicas).build();
    }

    /** Same partition count as the topic: a dead record keeps its partition number. */
    @Bean
    NewTopic paymentEventsDlt(@Value("${paycore.history.topic-partitions:3}") int partitions,
                              @Value("${paycore.history.topic-replicas:1}") int replicas) {
        return TopicBuilder.name(DLT).partitions(partitions).replicas(replicas).build();
    }

    @Bean
    DefaultErrorHandler errorHandler(KafkaTemplate<String, String> kafka) {
        DeadLetterPublishingRecoverer toDlt = new DeadLetterPublishingRecoverer(kafka,
                (record, exception) -> new TopicPartition(DLT, record.partition()));
        DefaultErrorHandler handler = new DefaultErrorHandler(toDlt,
                new FixedBackOff(1_000L, FixedBackOff.UNLIMITED_ATTEMPTS));
        handler.addNotRetryableExceptions(InvalidEventException.class);
        return handler;
    }
```
